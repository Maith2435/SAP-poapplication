namespace poapplication.common;
using{ Country,Currency} from '@sap/cds/common';
type nameStr : String(50);
aspect address{
    drno : String(20);
    street : nameStr;
    landmark : nameStr;
    city : String(100);
    postal: Integer;
    state :String(100);
    country :Country;
    region : String(20);
}
type gender : String(10) enum{
    M='Male';
    F='Female';
    U='Udisclosed'
}

type BookStatus : String enum {
Submitted = 'SUBMITTED';
Approved = 'APPROVED';
Issued = 'ISSUED';
Rejected = 'REJECTED';
}

type AmountT : Decimal(10,2) @(
    Semantics.amount.currencyCode : 'CURRENCY_CODE',
    sap.unit : 'CURRENCY_CODE'
);

type PhoneNumber : String(20)
@assert.format: '^(?:(?:\+|00)?91[\-\s]?)?[6-9]\d{9}$';

type Email : String(100)
@assert.format: '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$';

aspect feeasp{
    gross_fee : AmountT;
    tax:AmountT;
    total_fee : AmountT;
    currency : Currency;
}
